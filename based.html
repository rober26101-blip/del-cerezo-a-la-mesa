CREATE TABLE dbo.Categoria (
    id_Categoria    INT           IDENTITY(1,1)  NOT NULL,
    nombre_Categoria VARCHAR(100)               NOT NULL,
    descripcion     VARCHAR(255)               NULL,
    CONSTRAINT PK_Categoria PRIMARY KEY (id_Categoria),
    CONSTRAINT UQ_Categoria_Nombre UNIQUE (nombre_Categoria)
);
GO


IF OBJECT_ID('dbo.Producto', 'U') IS NOT NULL
    DROP TABLE dbo.Producto;
GO

CREATE TABLE dbo.Producto (
    id_Producto     INT           IDENTITY(1,1)  NOT NULL,
    nombre_Producto VARCHAR(150)               NOT NULL,
    descripcion     VARCHAR(255)               NULL,
    Precio          DECIMAL(10,2)              NOT NULL,
    stock           INT                        NOT NULL  DEFAULT 0,
    id_Categoria    INT                        NOT NULL,
    CONSTRAINT PK_Producto   PRIMARY KEY (id_Producto),
    CONSTRAINT FK_Producto_Categoria FOREIGN KEY (id_Categoria)
        REFERENCES dbo.Categoria (id_Categoria),
    CONSTRAINT CK_Producto_Precio CHECK (Precio >= 0),
    CONSTRAINT CK_Producto_Stock  CHECK (stock  >= 0)
);
GO


IF OBJECT_ID('dbo.Cliente', 'U') IS NOT NULL
    DROP TABLE dbo.Cliente;
GO

CREATE TABLE dbo.Cliente (
    id_Cliente      INT           IDENTITY(1,1)  NOT NULL,
    nombre          VARCHAR(100)               NOT NULL,
    apellido        VARCHAR(100)               NOT NULL,
    telefono        VARCHAR(20)                NULL,
    email           VARCHAR(150)               NULL,
    barrio          VARCHAR(100)               NULL,
    fecha_Registro  DATE                       NOT NULL  DEFAULT GETDATE(),
    CONSTRAINT PK_Cliente    PRIMARY KEY (id_Cliente),
    CONSTRAINT UQ_Cliente_Email UNIQUE (email)
);
GO


IF OBJECT_ID('dbo.Pedido', 'U') IS NOT NULL
    DROP TABLE dbo.Pedido;
GO

CREATE TABLE dbo.Pedido (
    id_Pedido       INT           IDENTITY(1,1)  NOT NULL,
    id_Cliente      INT                        NOT NULL,
    fecha_Pedido    DATETIME                   NOT NULL  DEFAULT GETDATE(),
    total           DECIMAL(12,2)              NOT NULL  DEFAULT 0,
    estado          VARCHAR(50)                NOT NULL  DEFAULT 'Pendiente',
    CONSTRAINT PK_Pedido     PRIMARY KEY (id_Pedido),
    CONSTRAINT FK_Pedido_Cliente FOREIGN KEY (id_Cliente)
        REFERENCES dbo.Cliente (id_Cliente),
    CONSTRAINT CK_Pedido_Estado CHECK (estado IN ('Pendiente','En preparacion','Entregado','Cancelado')),
    CONSTRAINT CK_Pedido_Total  CHECK (total  >= 0)
);
GO


IF OBJECT_ID('dbo.Detalle_Pedido', 'U') IS NOT NULL
    DROP TABLE dbo.Detalle_Pedido;

CREATE TABLE dbo.Detalle_Pedido (
    id_Detalle      INT           IDENTITY(1,1)  NOT NULL,
    id_Pedido       INT                        NOT NULL,
    id_Producto     INT                        NOT NULL,
    cantidad        INT                        NOT NULL,
    precio_unitario DECIMAL(10,2)              NOT NULL,
    subtotal        AS (cantidad * precio_unitario) PERSISTED,
    CONSTRAINT PK_Detalle    PRIMARY KEY (id_Detalle),
    CONSTRAINT FK_Detalle_Pedido   FOREIGN KEY (id_Pedido)
        REFERENCES dbo.Pedido   (id_Pedido),
    CONSTRAINT FK_Detalle_Producto FOREIGN KEY (id_Producto)
        REFERENCES dbo.Producto (id_Producto),
    CONSTRAINT CK_Detalle_Cantidad CHECK (cantidad > 0),
    CONSTRAINT CK_Detalle_PrecioU  CHECK (precio_unitario >= 0)
);
GO

--  Detalle completo: pedido, producto y categoria --
SELECT
    P.id_Pedido,
    C.nombre + ' ' + C.apellido    AS Cliente,
    PR.nombre_Producto,
    CAT.nombre_Categoria,
    D.cantidad,
    D.precio_unitario,
    D.subtotal
FROM dbo.Detalle_Pedido D
INNER JOIN dbo.Pedido   P   ON P.id_Pedido    = D.id_Pedido

INNER JOIN dbo.Producto PR  ON PR.id_Producto = D.id_Producto
INNER JOIN dbo.Categoria CAT ON CAT.id_Categoria = PR.id_Categoria
ORDER BY P.id_Pedido, D.id_Detalle;
GO

--Productos mas vendidos (TOP 5) --
SELECT TOP 5
    PR.nombre_Producto,
    CAT.nombre_Categoria,
    SUM(D.cantidad)   AS Total_Unidades_Vendidas,
    SUM(D.subtotal)   AS Total_Ingresos
FROM dbo.Detalle_Pedido D
INNER JOIN dbo.Producto  PR  ON PR.id_Producto   = D.id_Producto
INNER JOIN dbo.Categoria CAT ON CAT.id_Categoria = PR.id_Categoria
GROUP BY PR.id_Producto, PR.nombre_Producto, CAT.nombre_Categoria
ORDER BY Total_Unidades_Vendidas DESC;
GO

-- Clientes con mayor gasto total --

SELECT
    C.id_Cliente,
    C.nombre + ' ' + C.apellido  AS Cliente,
    C.barrio,
    COUNT(P.id_Pedido)            AS Num_Pedidos,
    SUM(P.total)                  AS Gasto_Total
FROM dbo.Cliente C
INNER JOIN dbo.Pedido P ON P.id_Cliente = C.id_Cliente
GROUP BY C.id_Cliente, C.nombre, C.apellido, C.barrio
ORDER BY Gasto_Total DESC;

IF TYPE_ID('dbo.TipoItemPedido') IS NOT NULL
    DROP TYPE dbo.TipoItemPedido;
GO

CREATE TYPE dbo.TipoItemPedido AS TABLE (
    id_Producto     INT           NOT NULL,
    cantidad        INT           NOT NULL
);
GO

IF OBJECT_ID('dbo.sp_RegistrarPedido', 'P') IS NOT NULL
    DROP PROCEDURE dbo.sp_RegistrarPedido;
GO

CREATE PROCEDURE dbo.sp_RegistrarPedido
    @id_Cliente     INT,
    @items          dbo.TipoItemPedido READONLY,
    @id_Pedido_OUT  INT           OUTPUT,
    @total_OUT      DECIMAL(12,2) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    -- Valido que el cliente existe --

    IF NOT EXISTS (SELECT 1 FROM dbo.Cliente WHERE id_Cliente = @id_Cliente)
    BEGIN
        RAISERROR('Cliente no encontrado.', 16, 1);
        RETURN;
    END

    -- Validar que hay items --
    IF NOT EXISTS (SELECT 1 FROM @items)
    BEGIN
        RAISERROR('El pedido debe tener al menos un producto.', 16, 1);
        RETURN;
    END

    -- Validar stock de cada producto --
    IF EXISTS (
        SELECT 1
        FROM @items I
        INNER JOIN dbo.Producto P ON P.id_Producto = I.id_Producto
        WHERE P.stock < I.cantidad
    )
    BEGIN
        RAISERROR('Stock insuficiente para uno o mas productos.', 16, 1);
        RETURN;
    END

    BEGIN TRANSACTION;
    BEGIN TRY
        -- Insertar cabecera del pedido --

        INSERT INTO dbo.Pedido (id_Cliente, fecha_Pedido, total, estado)
        VALUES (@id_Cliente, GETDATE(), 0, 'Pendiente');

        SET @id_Pedido_OUT = SCOPE_IDENTITY();

        -- Insertar detalles tomando precio actual del producto --
        INSERT INTO dbo.Detalle_Pedido (id_Pedido, id_Producto, cantidad, precio_unitario)
        SELECT
            @id_Pedido_OUT,
            I.id_Producto,
            I.cantidad,
            P.Precio
        FROM @items I
        INNER JOIN dbo.Producto P ON P.id_Producto = I.id_Producto;

        -- Descontar stock --
        UPDATE P
        SET    P.stock = P.stock - I.cantidad
        FROM   dbo.Producto P
        INNER JOIN @items I ON I.id_Producto = P.id_Producto;

        -- Calcular y actualizar total del pedido
        SELECT @total_OUT = SUM(subtotal)
        FROM   dbo.Detalle_Pedido
        WHERE  id_Pedido = @id_Pedido_OUT;

        UPDATE dbo.Pedido
        SET    total = @total_OUT
        WHERE  id_Pedido = @id_Pedido_OUT;

        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        ROLLBACK TRANSACTION;
        THROW;
    END CATCH
END;
GO


  /* Actualizar estado de un pedido
 IN:  id_pedido, nuevo_estado
 OUT: mensaje de confirmacion */

IF OBJECT_ID('dbo.sp_ActualizarEstadoPedido', 'P') IS NOT NULL
    DROP PROCEDURE dbo.sp_ActualizarEstadoPedido;
GO

CREATE PROCEDURE dbo.sp_ActualizarEstadoPedido
    @id_Pedido      INT,
    @nuevo_estado   VARCHAR(50),
    @mensaje_OUT    VARCHAR(200) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    IF NOT EXISTS (SELECT 1 FROM dbo.Pedido WHERE id_Pedido = @id_Pedido)
    BEGIN
        SET @mensaje_OUT = 'ERROR: Pedido ' + CAST(@id_Pedido AS VARCHAR) + ' no existe.';
        RETURN;
    END

    IF @nuevo_estado NOT IN ('Pendiente','En preparacion','Entregado','Cancelado')
    BEGIN
        SET @mensaje_OUT = 'ERROR: Estado "' + @nuevo_estado + '" no es valido.';
        RETURN;
    END

    DECLARE @estado_actual VARCHAR(50);
    SELECT @estado_actual = estado FROM dbo.Pedido WHERE id_Pedido = @id_Pedido;

    IF @estado_actual = 'Entregado'
    BEGIN
        SET @mensaje_OUT = 'ADVERTENCIA: El pedido ya fue entregado y no puede modificarse.';
        RETURN;
    END

    UPDATE dbo.Pedido
    SET    estado = @nuevo_estado
    WHERE  id_Pedido = @id_Pedido;

    SET @mensaje_OUT = 'OK: Pedido ' + CAST(@id_Pedido AS VARCHAR) +
                       ' actualizado de "' + @estado_actual +
                       '" a "' + @nuevo_estado + '".';
END;
GO

CREATE PROCEDURE dbo.sp_HistorialCliente
    @id_Cliente         INT,
    @num_pedidos_OUT    INT           OUTPUT,
    @total_gastado_OUT  DECIMAL(12,2) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    IF NOT EXISTS (SELECT 1 FROM dbo.Cliente WHERE id_Cliente = @id_Cliente)
    BEGIN
        RAISERROR('Cliente no encontrado.', 16, 1);
        RETURN;
    END

    
    SELECT
        @num_pedidos_OUT   = COUNT(*),
        @total_gastado_OUT = ISNULL(SUM(total), 0)
    FROM dbo.Pedido
    WHERE id_Cliente = @id_Cliente;

    /* Resultado detallado */

    SELECT
        P.id_Pedido,
        P.fecha_Pedido,
        P.estado,
        P.total,
        (
            SELECT STRING_AGG(PR.nombre_Producto + ' x' + CAST(D.cantidad AS VARCHAR), ', ')
            FROM   dbo.Detalle_Pedido D
            INNER JOIN dbo.Producto PR ON PR.id_Producto = D.id_Producto
            WHERE  D.id_Pedido = P.id_Pedido
        ) AS Productos
    FROM dbo.Pedido P
    WHERE P.id_Cliente = @id_Cliente
    ORDER BY P.fecha_Pedido DESC;
END;
GO

/* reporte de ventas por rango de fecha */

IF OBJECT_ID('dbo.sp_ReporteVentasCategorias', 'P') IS NOT NULL
    DROP PROCEDURE dbo.sp_ReporteVentasCategorias;
GO

CREATE PROCEDURE dbo.sp_ReporteVentasCategorias
    @fecha_inicio       DATE,
    @fecha_fin          DATE,
    @num_categorias_OUT INT OUTPUT
AS
BEGIN
    SET NOCOUNT ON;

    IF @fecha_inicio > @fecha_fin
    BEGIN
        RAISERROR('La fecha de inicio no puede ser mayor a la fecha fin.', 16, 1);
        RETURN;
    END

    SELECT
        CAT.nombre_Categoria,
        COUNT(DISTINCT P.id_Pedido)   AS Num_Pedidos,
        SUM(D.cantidad)               AS Unidades_Vendidas,
        SUM(D.subtotal)               AS Ingresos_Total,
        AVG(D.subtotal)               AS Ticket_Promedio
    FROM dbo.Detalle_Pedido D
    INNER JOIN dbo.Pedido   P   ON P.id_Pedido    = D.id_Pedido
    INNER JOIN dbo.Producto PR  ON PR.id_Producto = D.id_Producto
    INNER JOIN dbo.Categoria CAT ON CAT.id_Categoria = PR.id_Categoria
    WHERE CAST(P.fecha_Pedido AS DATE) BETWEEN @fecha_inicio AND @fecha_fin
    GROUP BY CAT.id_Categoria, CAT.nombre_Categoria
    ORDER BY Ingresos_Total DESC;

    SELECT @num_categorias_OUT = @@ROWCOUNT;
END;
GO
