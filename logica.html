public class LoginDelCerezo { // Constantes de credenciales y límites private static final String LOGIN_CORRECTO = "code"; private static final String CLAVE_CORRECTA = "123"; private static final int MAX_INTENTOS = 3; public static void main(String[] args) { Scanner scanner = new Scanner(System.in); int intentos = 0; // Contador de intentos (equivalente a tu variable 'i') boolean accesoConcedido = false; System.out.println(" "); System.out.println("SISTEMA DE CONTROL DE ACCESO"); System.out.println("DEL CEREZO A LA MESA"); System.out.println(" "); // El bucle se ejecuta mientras no se superen los intentos y no haya ingresado while (intentos < MAX_INTENTOS && !accesoConcedido) { System.out.print("\nIngrese su usuario: "); String loginInput = scanner.nextLine().trim(); System.out.print("Ingrese su contraseña: "); String claveInput = scanner.nextLine().trim(); // Validación de credenciales if (loginInput.equals(LOGIN_CORRECTO) && claveInput.equals(CLAVE_CORRECTA)) { accesoConcedido = true; } else { intentos++; // Incrementar contador de intentos fallidos marcarIntentoFallido(intentos); } } // 4. Manejo del resultado final del login evaluarResultadoLogin(accesoConcedido); scanner.close(); } /** * Método equivalente a 'marcarIntento(i)' en tu JavaScript. * Muestra visualmente el estado del error según el intento actual. */ private static void marcarIntentoFallido(int intentoActual) { if (intentoActual < MAX_INTENTOS) { System.out.println("\n "); System.out.printf("❌ Clave incorrecta. Intento %d de %d.%n", intentoActual, MAX_INTENTOS); System.out.println(" "); } } /** * Procesa y muestra el desenlace final del intento de login. */ private static void evaluarResultadoLogin(boolean exito) { System.out.println("\n "); if (exito) { System.out.println("✅ Bienvenido al programa."); System.out.println("Acceso concedido exitosamente."); // Aquí podrías llamar a tu método mostrarBienvenida() o iniciar la App de ventas } else { System.out.println("🔒 Sistema bloqueado."); System.out.println("Demasiados intentos fallidos de inicio de sesión."); } System.out.println(" "); } }

import java.util.Scanner; public class DelCerezoALaMesa { // Constantes para los precios de los productos private static final int PRECIO_MERMELADA = 18000; private static final int PRECIO_DULCES = 12000; public static void main(String[] args) { Scanner scanner = new Scanner(System.in); System.out.println(" "); System.out.println(" BIENVENIDO A: DEL CEREZO A LA MESA "); System.out.println(" "); // 1. Solicitar cantidad de Mermeladas System.out.println("\nProducto 1: Mermelada Exótica (Frasco de 250g)"); System.out.println("Precio unitario: $" + PRECIO_MERMELADA + " COP"); System.out.print("¿Cuántas unidades desea llevar?: "); int cantMermelada = leerCantidad(scanner); // 2. Solicitar cantidad de Dulces Artesanales System.out.println("\nProducto 2: Dulces Artesanales (Paquete de 150g)"); System.out.println("Precio unitario: $" + PRECIO_DULCES + " COP"); System.out.print("¿Cuántas unidades desea llevar?: "); int cantDulces = leerCantidad(scanner); // 3. Realizar los cálculos de la venta int subtotalMermelada = cantMermelada * PRECIO_MERMELADA; int subtotalDulces = cantDulces * PRECIO_DULCES; int totalPagar = subtotalMermelada + subtotalDulces; // 4. Generar el recibo de venta if (totalPagar > 0) { generarRecibo(cantMermelada, subtotalMermelada, cantDulces, subtotalDulces, totalPagar); } else { System.out.println("\nNo se realizó ninguna venta. ¡Gracias por visitarnos!"); } scanner.close(); } /** * Método auxiliar para validar que el usuario ingrese un número entero positivo */ private static int leerCantidad(Scanner scanner) { while (!scanner.hasNextInt()) { System.out.println("Por favor, ingresa un número válido."); scanner.next(); } int cantidad = scanner.nextInt(); return Math.max(cantidad, 0); // Evita que ingresen números negativos } /** * Método para imprimir el formato del recibo de pago */ private static void generarRecibo(int cantMermelada, int subMermelada, int cantDulces, int subDulces, int total) { System.out.println("\n "); System.out.println(" FACTURA DE VENTA "); System.out.println(" "); if (cantMermelada > 0) { System.out.printf("- Mermelada Exótica (250g) x%d : $%d COP%n", cantMermelada, subMermelada); } if (cantDulces > 0) { System.out.printf("- Dulces Artesanales (150g) x%d: $%d COP%n", cantDulces, subDulces); } System.out.println(" "); System.out.printf("TOTAL A PAGAR: $%d COP%n", total); System.out.println(" "); System.out.println(" ¡Gracias por apoyar la sostenibilidad!"); System.out.println(" "); } }
