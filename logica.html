       import java.util.Scanner;

class Producto {
    String nombreProducto;
    double precio;
    int stock;

    public Producto(String nombreProducto, double precio, int stock) {
        this.nombreProducto = nombreProducto;
        this.precio = precio;
        this.stock = stock;
    }
}

public class DelCerezoALaMesa {
    public static void main(String[] args) {
        Scanner teclado = new Scanner(System.in);

        Producto mermelada = new Producto("Mermelada Exótica 250g", 18000, 50);
        Producto dulce = new Producto("Dulce Artesanal 150g", 12000, 80);

        System.out.println("--- BIENVENIDO A DEL CEREZO A LA MESA ---");
        System.out.println("Productos disponibles:");
        System.out.println("1. " + mermelada.nombreProducto + " - $" + mermelada.precio + " (Stock: " + mermelada.stock + ")");
        System.out.println("2. " + dulce.nombreProducto + " - $" + dulce.precio + " (Stock: " + dulce.stock + ")");
        
        
        System.out.print("\nSelecciona el número del producto que deseas comprar: ");
        int opcion = teclado.nextInt();
        
        System.out.print("¿Cuántas unidades deseas llevar?: ");
        int cantidadSolicitada = teclado.nextInt();

        Producto productoSeleccionado = null;

        if (opcion == 1) {
            productoSeleccionado = mermelada;
        } else if (opcion == 2) {
            productoSeleccionado = dulce;
        } else {
            System.out.println("Opción no válida.");
        }

        if (productoSeleccionado != null) {
            if (cantidadSolicitada <= productoSeleccionado.stock) {
                // Cálculo del subtotal
                double subtotal = cantidadSolicitada * productoSeleccionado.precio;
                
                
                productoSeleccionado.stock -= cantidadSolicitada;

                System.out.println("\n--- RESUMEN DE COMPRA ---");
                System.out.println("Producto: " + productoSeleccionado.nombreProducto);
                System.out.println("Cantidad: " + cantidadSolicitada);
                System.out.println("Total a pagar: $" + subtotal);
                System.out.println("Estado del Pedido: Pendiente");
                System.out.println("Nuevo Stock disponible: " + productoSeleccionado.stock);
            } else {
                System.out.println("\n[ERROR] No se pudo procesar el pedido.");
                System.out.println("Motivo: Stock insuficiente. Solo quedan " + productoSeleccionado.stock + " unidades.");
            }
        }
    }
}

CREATE DATABASE del_cerezo_a_la_mesa; USE del_cerezo_a_la_mesa; -- Tabla Categoria CREATE TABLE Categoria ( id_Categoria INT PRIMARY KEY AUTO_INCREMENT, nombre_Categoria VARCHAR(80) NOT NULL, descripcion TEXT ); CREATE TABLE Cliente ( id_Cliente INT PRIMARY KEY AUTO_INCREMENT, nombre VARCHAR(60) NOT NULL, apellido VARCHAR(60) NOT NULL, telefono VARCHAR(15), email VARCHAR(100) UNIQUE NOT NULL, barrio VARCHAR(60), fecha_Registro DATE DEFAULT (CURRENT_DATE) ); -- Tabla Producto CREATE TABLE Producto ( id_Producto INT PRIMARY KEY AUTO_INCREMENT, nombre_Producto VARCHAR(100) NOT NULL, descripcion TEXT, Precio DECIMAL(10,2) NOT NULL, stock INT DEFAULT 0, id_Categoria INT, FOREIGN KEY (id_Categoria) REFERENCES Categoria(id_Categoria) ); CREATE TABLE Pedido ( id_Pedido INT PRIMARY KEY AUTO_INCREMENT, id_Cliente INT NOT NULL, fecha_Pedido DATETIME DEFAULT CURRENT_TIMESTAMP, total DECIMAL(10,2), estado ENUM('pendiente', 'enviado', 'entregado') DEFAULT 'pendiente', FOREIGN KEY (id_Cliente) REFERENCES Cliente(id_Cliente) ) CREATE TABLE Detalle ( id_Detalle INT PRIMARY KEY AUTO_INCREMENT, id_Pedido INT NOT NULL, id_producto INT NOT NULL, cantidad INT NOT NULL, precio_unitario DECIMAL(10,2) NOT NULL, subtotal DECIMAL(10,2) NOT NULL, FOREIGN KEY (id_Pedido) REFERENCES Pedido(id_Pedido), FOREIGN KEY (id_producto) REFERENCES Producto(id_Producto) );
